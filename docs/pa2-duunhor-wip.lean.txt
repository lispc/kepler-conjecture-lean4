/-- 凸包保持仿射维数（本波自证；`ROGERS_AFF_DIM_FULL` 内含同型论证）。 -/
private theorem p2g_affDim_convexHull (s : Set V3) :
    affDim (convexHull ℝ s) = affDim s := by
  by_cases hs : s = ∅
  · rw [hs, convexHull_empty, affDim_empty]
  · have hsne : (convexHull ℝ s) ≠ ∅ := by
      obtain ⟨x, hx⟩ := Set.nonempty_iff_ne_empty.2 hs
      intro hcon
      exact Set.notMem_empty x (hcon ▸ subset_convexHull ℝ s hx)
    have hs' : s.Nonempty := Set.nonempty_iff_ne_empty.2 hs
    refine le_antisymm ?_ ?_
    · have hmono : vectorSpan ℝ ↑(convexHull ℝ s) ≤ vectorSpan ℝ s := by
        have hstep : vectorSpan ℝ ↑(convexHull ℝ s) ≤
            vectorSpan ℝ ↑(affineSpan ℝ s) :=
          vectorSpan_mono ℝ (convexHull_subset_affineSpan s)
        rw [← AffineSubspace.direction_eq_vectorSpan (affineSpan ℝ s), direction_affineSpan] at hstep
        exact hstep
      rw [affDim, if_neg hsne, affDim, if_neg (nonempty_iff_ne_empty.mp hs')]
      exact Nat.cast_le.2 (Submodule.finrank_mono hmono)
    · rw [affDim, if_neg (nonempty_iff_ne_empty.mp hs'), affDim, if_neg hsne]
      exact Nat.cast_le.2
        (Submodule.finrank_mono (vectorSpan_mono ℝ (subset_convexHull ℝ s)))

        rw [heq]; abel
      linear_combination (norm := module) e1
    have h2 : s • q = s • ((1 - t / s) • p0 + (t / s) • p) := by

      have hB : s * (t / s) = t := by field_simp
      rw [hA, hB]

  have hseg : q ∈ openSegment ℝ p0 p := by
    rw [hq2]
    simp only [openSegment, Set.mem_setOf_eq]

    rw [sub_pos, div_lt_one hs]

  obtain ⟨h1fo, -, -⟩ := h1
  have hpP : p ∈ P := h1fo.1 hp
  obtain ⟨ha, -⟩ := h2fo.2.2 p0 p q hp0 hpP hq hseg

    exact hx'
